# CSE 422: Advanced Distributed Systems Architecture
# Coroutine task scheduling and non-blocking I/O primitives

import asyncio
import sys
import logging

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")

class AsyncPipelineWorker:
    def __init__(self, worker_id: int, queue: asyncio.Queue):
        self.worker_id = worker_id
        self.queue = queue
        self.is_running = True

    async def process_work_unit(self, task_payload: dict) -> bool:
        task_id = task_payload.get("uuid")
        logging.info(f"Processing execution unit {task_id} on thread loop")
        await asyncio.sleep(0.045) # emulate non-blocking socket stream
        return True

    async def run(self):
        while self.is_running:
            payload = await self.queue.get()
            await self.process_work_unit(payload)
            self.queue.task_done()

async def main():
    task_queue = asyncio.Queue(maxsize=1024)
    workers = [AsyncPipelineWorker(i, task_queue) for i in range(8)]
    runner_group = asyncio.gather(*(w.run() for w in workers))
    await task_queue.join()

if __name__ == "__main__":
    asyncio.run(main())
